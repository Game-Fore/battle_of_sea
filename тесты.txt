# ----------------------------------------
# TCP JSON тест для сервера морского боя
# ----------------------------------------

# Подключаемся к серверу
$client = New-Object System.Net.Sockets.TcpClient("127.0.0.1",5000)
$stream = $client.GetStream()

# Создаем writer и reader
$writer = New-Object System.IO.StreamWriter($stream,[System.Text.Encoding]::UTF8)
$writer.AutoFlush = $true
$reader = New-Object System.IO.StreamReader($stream,[System.Text.Encoding]::UTF8)

# Функция отправки JSON и получения ответа
function Send-Json($json) {
    $writer.Write("$json`n")  # Обязательно \n для ReadLineAsync()
    $response = $reader.ReadLine()
    Write-Host "Server response: $response"
}

#1️⃣ Отправляем connect
$playerName = "Alex"
$connectJson = "{""type"":""connect"",""payload"":{""playerName"":""$playerName""}}"
Write-Host "Sending: $connectJson"
Send-Json $connectJson

# 2️⃣ Отправляем ping
$pingJson = '{"type":"ping","payload":{}}'
Write-Host "Sending: $pingJson"
Send-Json $pingJson

# 3️⃣ Закрываем соединение
$writer.Close()
$reader.Close()
$client.Close()
Write-Host "Connection closed"
